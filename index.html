  <script>

    const watchData = {

      champ: [

        { name: "Active Runner", img: "Frame 2147226976.png" },

        { name: "Monkey Bar Boy", img: "Frame 2147226978.png" },

        { name: "High Five Hands", img: "Packaging watchface.png" },

        { name: "Lego Bricks", img: "Lego (1).png" }

      ],

      explorer: [

        { name: "Active Runner", img: "Frame 2147226976.png" },

        { name: "Window Cat", img: "Gemini_Generated_Image_1ppnm1ppnm1ppnm1.jpeg" },

        { name: "Sneaker Step", img: "Gemini_Generated_Image_1ppnm1ppnm1ppnm1 (1).jpeg" },

        { name: "Lazy Capybara", img: "Capybara.png" }

      ]

    };

    let currentTab = 'champ';

    let currentPuzzleIndex = 0;

    let placedCount = 0;

    const boardSlots = document.querySelectorAll('.board-slot');

    const piecesPanel = document.getElementById('pieces-panel');

    const faceNameEl = document.getElementById('face-name');

    const modal = document.getElementById('win-modal');

    function loadPuzzle() {

      placedCount = 0;

      modal.style.display = 'none';

      const activeItem = watchData[currentTab][currentPuzzleIndex];

      faceNameEl.textContent = activeItem.name;

      // Clear Board Slots

      boardSlots.forEach((slot, i) => {

        slot.innerHTML = '';

        slot.classList.remove('filled');

        slot.style.backgroundImage = 'none';

      });

      // Clear & Build Pieces

      piecesPanel.innerHTML = '';

     

      // Generate 6 slices (3 columns x 2 rows)

      let pieces = [];

      for (let row = 0; row < 2; row++) {

        for (let col = 0; col < 3; col++) {

          const index = row * 3 + col;

          pieces.push({ index, row, col });

        }

      }

      // Shuffle pieces

      pieces.sort(() => Math.random() - 0.5);

      pieces.forEach(p => {

        const piece = document.createElement('div');

        piece.className = 'puzzle-piece';

        piece.draggable = true;

        piece.dataset.index = p.index;

       

        // Render slice portion accurately

        piece.style.backgroundImage = `url('${activeItem.img}')`;

        piece.style.backgroundPosition = `${p.col * 50}% ${p.row * 100}%`;

        piece.addEventListener('dragstart', (e) => {

          e.dataTransfer.setData('text/plain', p.index);

        });

        piecesPanel.appendChild(piece);

      });

    }

    // Set up slot dragover and drop handlers

    boardSlots.forEach(slot => {

      slot.addEventListener('dragover', (e) => e.preventDefault());

     

      slot.addEventListener('drop', (e) => {

        e.preventDefault();

        const draggedIndex = e.dataTransfer.getData('text/plain');

        const slotIndex = slot.dataset.index;

        if (draggedIndex === slotIndex && !slot.classList.contains('filled')) {

          slot.classList.add('filled');

         

          // Place image slice smoothly on the board

          const activeItem = watchData[currentTab][currentPuzzleIndex];

          const col = slotIndex % 3;

          const row = Math.floor(slotIndex / 3);

         

          slot.style.backgroundImage = `url('${activeItem.img}')`;

          slot.style.backgroundPosition = `${col * 50}% ${row * 100}%`;

          // Hide corresponding piece from panel

          const pieceEl = piecesPanel.querySelector(`[data-index="${draggedIndex}"]`);

          if (pieceEl) pieceEl.remove();

          placedCount++;

          if (placedCount === 6) {

            setTimeout(() => {

              modal.style.display = 'flex';

            }, 300);

          }

        }

      });

    });

    // Tab Event Listeners

    document.getElementById('tab-champ').addEventListener('click', (e) => {

      currentTab = 'champ';

      currentPuzzleIndex = 0;

      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));

      e.target.classList.add('active');

      loadPuzzle();

    });

    document.getElementById('tab-explorer').addEventListener('click', (e) => {

      currentTab = 'explorer';

      currentPuzzleIndex = 0;

      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));

      e.target.classList.add('active');

      loadPuzzle();

    });

    function nextPuzzle() {

      currentPuzzleIndex = (currentPuzzleIndex + 1) % watchData[currentTab].length;

      loadPuzzle();

    }

    // Initialize initial puzzle load

    loadPuzzle();

  </script>
